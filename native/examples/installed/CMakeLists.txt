cmake_minimum_required(VERSION 3.18)
project(parsegraph_consumer LANGUAGES C CXX)

find_package(parsegraph 0.1 CONFIG REQUIRED)
add_executable(graph_c graph.c)
target_link_libraries(graph_c PRIVATE parsegraph::parsegraph)
add_executable(graph_cpp graph.cpp)
target_compile_features(graph_cpp PRIVATE cxx_std_11)
target_link_libraries(graph_cpp PRIVATE parsegraph::parsegraph)
add_executable(extent_c extent.c)
target_link_libraries(extent_c PRIVATE parsegraph::extent)
enable_testing()
add_test(NAME graph_c COMMAND graph_c)
add_test(NAME graph_cpp COMMAND graph_cpp)
add_test(NAME extent_c COMMAND extent_c)

# Acceptance-only checks; normal users need only find_package and target_link_libraries.
if(EXPECTED_PREFIX)
  foreach(target parsegraph::parsegraph parsegraph::extent)
    foreach(property INTERFACE_COMPILE_OPTIONS INTERFACE_LINK_OPTIONS)
      get_target_property(value ${target} ${property})
      if(value)
        message(FATAL_ERROR "Unexpected consumer flags: ${target} ${property}=${value}")
      endif()
    endforeach()
    get_target_property(configs ${target} IMPORTED_CONFIGURATIONS)
    foreach(config IN LISTS configs)
      get_target_property(archive ${target} IMPORTED_LOCATION_${config})
      string(FIND "${archive}" "${EXPECTED_PREFIX}/" prefix_position)
      if(NOT prefix_position EQUAL 0 OR NOT EXISTS "${archive}")
        message(FATAL_ERROR "Archive escaped the relocated package: ${archive}")
      endif()
    endforeach()
  endforeach()
endif()
